// ==========================================
// 3. MOTOR TD + FIBONACCI (función autónoma)
//    Devuelve: [señal, estado, entrada, sl, tp1, tp2, lote1, lote2, tradesHoy]
//    señal: 1 buy · -1 sell · 2 cancelación · 0 nada
//    estado: ±3 señal · 4 máx. trades · ±2 setup · ±1 impulso · 0 nada
// ==========================================
f_engine(bool bloqueado) =>
    var float ini         = na
    var float fin         = na
    var int   dir         = 0
    var int   velaAct     = na
    var int   tradesHoy   = 0
    var int   ultimaSenal = na

    bool ventana  = not na(time(timeframe.period, sessionOper, useTimeZone))
    bool nuevoDia = time("D") != time("D")[1]

    if nuevoDia
        tradesHoy := 0
        dir       := 0
        velaAct   := na
    if not ventana
        dir     := 0
        velaAct := na

    bool tdDemand = low[2] < low[1] and low[2] < low[3] and low[2] < low and low[2] < low[4]
    bool tdSupply = high[2] > high[1] and high[2] > high[3] and high[2] > high and high[2] > high[4]

    bool cooldownOK  = na(ultimaSenal) or (bar_index - ultimaSenal >= cooldownVelas)
    bool puedeOperar = ventana and tradesHoy < maxTradesDia and not bloqueado and tfOK and cooldownOK

    // 1) Completar impulsos pendientes
    if tdSupply and dir == 1 and puedeOperar
        fin := high[2]
        if (fin - ini) >= minImpulso
            dir     := 2
            velaAct := bar_index
        else
            dir := 0
    if tdDemand and dir == -1 and puedeOperar
        fin := low[2]
        if (ini - fin) >= minImpulso
            dir     := -2
            velaAct := bar_index
        else
            dir := 0

    // 2) Iniciar / actualizar impulsos (sin destruir un setup activo ±2)
    if tdDemand and puedeOperar and (dir == 0 or dir == 1)
        ini := low[2]
        dir := 1
    if tdSupply and puedeOperar and (dir == 0 or dir == -1)
        ini := high[2]
        dir := -1

    // 3) Niveles Fibonacci
    float entrada = na
    float sl      = na
    float tp1     = na
    float tp2     = na
    if dir == 2
        float rango = fin - ini
        entrada := fin - rango * fiboEntrada + bufferEntrada
        sl      := ini - spreadEnPrecio - ticksSL * syminfo.mintick
        tp1     := fin
        tp2     := fin + rango * fiboTP2
    if dir == -2
        float rango = ini - fin
        entrada := fin + rango * fiboEntrada - bufferEntrada
        sl      := ini + spreadEnPrecio + ticksSL * syminfo.mintick
        tp1     := fin
        tp2     := fin - rango * fiboTP2

    // 4) Lotes 60/40 (referenciales; el EA debe recalcular por instrumento)
    float distPips = math.abs(entrada - sl) / pipSize
    float lotes    = distPips > 0 ? capitalCuenta * pctRiesgo / (distPips * valorPipLote) : 0.0
    lotes := math.max(0.02, math.round(lotes, 2))
    float l1 = math.max(0.01, math.round(lotes * 0.60, 2))
    float l2 = math.max(0.01, math.round(lotes * 0.40, 2))

    // 5) Gatillos y expiración
    bool trigCompra = dir == 2 and low <= entrada and close > sl and puedeOperar
    bool trigVenta  = dir == -2 and high >= entrada and close < sl and puedeOperar
    bool expirado   = not na(velaAct) and (dir == 2 or dir == -2) and (bar_index - velaAct > expiraVelas) and not trigCompra and not trigVenta

    if trigCompra or trigVenta
        tradesHoy   += 1
        ultimaSenal := bar_index
        dir         := 0
        velaAct     := na
    if expirado
        dir     := 0
        velaAct := na

    int senal  = trigCompra ? 1 : trigVenta ? -1 : expirado ? 2 : 0
    int estado = trigCompra ? 3 : trigVenta ? -3 : tradesHoy >= maxTradesDia ? 4 : dir

    [senal, estado, entrada, sl, tp1, tp2, l1, l2, tradesHoy]

    array<string> partes = str.split(s, ":")
